**05.** 从有序顺序表中删除所有其值重复的元素，使表中所有元素的值均不同。

**答案：**

```cpp
void deleteElem(SqList &L)
{
    if (L.length == 0) return;

    int k = 0;
    for (int i = 0; i < L.length - 1; i++)
    {
        if (L.data[i] != L.data[i + 1])
        {
            L.data[k++] = L.data[i];
        }
    }

    L.data[k] = L.data[L.length - 1];
    L.length = k + 1;

    return;
}
```

```cpp
void Merge(SqList &a, SqList &b)
{
    if (a.length + b.length > MaxSize) return;
    int i = a.length - 1;
    int j = b.length - 1;
    int k = a.length + b.length - 1;

    while (i >= 0 && j >= 0)
    {
        if (a.data[i] >= b.data[j])
        {
            a.data[k--] = a.data[i--];
        }
        else
        {
            a.data[k--] = b.data[j--];
        }
    }

    while (j >= 0)
    {
        a.data[k--] = b.data[j--];
    }

    a.length = a.length + b.length;
}
```

```cpp
void reverse(ElemType *a, int left, int right)
{
    while (left < right)
    {
        ElemType temp = a[left];
        a[left++] = a[right];
        a[right--] = temp;
    }
}
```

**08.** 线性表 $(a_1,a_2,a_3,\ldots,a_n)$ 中的元素递增有序且按顺序存储于计算机内。要求设计一个算法，完成用最少时间在表中查找数值为 $x$ 的元素，若找到，则将其与后继元素位置相交换，若找不到，则将其插入表中并使表中元素仍然递增有序。

**答案：**

```cpp
void LocateElem(ElemType *a, int &n, ElemType x)
{
    int low = 0, high = n - 1, mid;
    while (low <= high)
    {
        mid = (high - low) / 2 + low;
        if (a[mid] == x) break;
        else if (a[mid] > x)
        {
            high = mid - 1;
        }
        else
        {
            low = mid + 1;
        }
    }

    if (low <= high && mid != n - 1 && a[mid] == x)
    {
        ElemType temp = a[mid];
        a[mid] = a[mid + 1];
        a[mid + 1] = temp;
    }

    if (low > high)
    {
        for (int i = n - 1; i > high; i--) a[i + 1] = a[i];
        a[high + 1] = x;
        n++;
    }
}
```
